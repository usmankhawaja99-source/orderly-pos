  <script>
    let activeFilter = 'all';
    let chosenMods = [];
    let activeTable = 'T1';

    const menu = [
      { id: 'burger', name: 'Double Smash Burger', price: 1450, tags: ['halal'], stock: 8 }, // Low stock warning (<15)
      { id: 'tacos', name: 'Birria Tacos (3pc)', price: 1600, tags: ['gf'], stock: 35 },
      { id: 'bowl', name: 'Harvest Quinoa Bowl', price: 1350, tags: ['gf', 'vg'], stock: 2 }, // Near zero
      { id: 'fries', name: 'Truffle Parm Fries', price: 750, tags: ['gf'], stock: 45 }
    ];

    const mods = [
      { id: 'm1', name: 'No Onions', price: 0 },
      { id: 'm2', name: 'Sub GF Bun', price: 200 },
      { id: 'm3', name: 'Extra Sauce', price: 100 }
    ];

    const tables = {
      T1: { id: 'T1', name: 'Table 1 (Patio)', items: [] },
      T2: { id: 'T2', name: 'Table 2 (Patio)', items: [] },
      T3: { id: 'T3', name: 'Table 3 (Window)', items: [] },
      T4: { id: 'T4', name: 'Table 4 (Booth)', items: [] }
    };

    let kds = [];
    let historyOrders = [
      { id: 'ord_901', channel: 'pos_terminal', total: 3872, stripeFee: 142, compFee: 151 }
    ];
    let lastReceiptHtml = '';

    function setFilter(tag, el) {
      activeFilter = tag;
      document.querySelectorAll('.filters .chip').forEach(c => c.classList.remove('active'));
      el.classList.add('active');
      renderMenu();
    }

    function toggleMod(id) {
      if (chosenMods.includes(id)) chosenMods = chosenMods.filter(m => m !== id);
      else if (chosenMods.length < 3) chosenMods.push(id);
      renderMenu();
    }

    function addItem(id) {
      const item = menu.find(i => i.id === id);
      if (item.stock <= 0) return alert('Item is out of stock!');
      item.stock -= 1;

      tables[activeTable].items.push({ id, mods: [...chosenMods] });
      chosenMods = [];
      renderAll();
    }

    function clearCheck() {
      tables[activeTable].items = [];
      renderAll();
    }

    function renderMenu() {
      document.getElementById('mod-count').textContent = `${chosenMods.length} / 3 Selected`;
      document.getElementById('mod-grid').innerHTML = mods.map(m => `
        <div class="mod-pill ${chosenMods.includes(m.id) ? 'selected' : ''}" onclick="toggleMod('${m.id}')">
          ${m.name} ${m.price ? `(+$${m.price/100})` : ''}
        </div>
      `).join('');

      const filtered = menu.filter(i => activeFilter === 'all' || i.tags.includes(activeFilter));
      document.getElementById('menu-grid').innerHTML = filtered.map(item => {
        const low = item.stock < 15 && item.stock > 0;
        const out = item.stock <= 0;
        return `
          <div class="tile ${out ? 'out-of-stock' : ''}" onclick="addItem('${item.id}')">
            <strong>${item.name}</strong>
            <div class="tile-price">$${(item.price/100).toFixed(2)}</div>
            <div class="tile-meta">
              ${item.tags.join(', ').toUpperCase()}<br/>
              <span style="color:${out ? 'var(--danger)' : low ? 'var(--warn)' : 'var(--accent)'}; font-weight:bold;">
                Stock: ${item.stock} ${low ? '(LOW!)' : ''}
              </span>
            </div>
          </div>
        `;
      }).join('');
    }

    function renderCheck() {
      const t = tables[activeTable];
      document.getElementById('current-table-title').textContent = t.name;
      document.getElementById('nav-t').textContent = activeTable;

      const listEl = document.getElementById('check-list');
      if (t.items.length === 0) {
        listEl.innerHTML = '<div style="color:var(--muted); text-align:center; padding:20px 0;">Check is empty</div>';
      } else {
        listEl.innerHTML = t.items.map((line, idx) => {
          const item = menu.find(i => i.id === line.id);
          const mNames = line.mods.map(mId => mods.find(m => m.id === mId)?.name).join(', ');
          const mPrice = line.mods.reduce((s, mId) => s + (mods.find(m => m.id === mId)?.price || 0), 0);
          const totalLine = item.price + mPrice;
          return `
            <div class="check-row">
              <div>
                <strong>1x ${item.name}</strong>
                ${mNames ? `<div style="font-size:11px; color:var(--accent);">+ ${mNames}</div>` : ''}
              </div>
              <div>
                $${(totalLine/100).toFixed(2)}
                <button onclick="tables['${activeTable}'].items.splice(${idx},1);renderAll();" style="background:none;border:none;color:var(--danger);cursor:pointer;margin-left:6px;">✕</button>
              </div>
            </div>
          `;
        }).join('');
      }

      let sub = 0;
      t.items.forEach(line => {
        const item = menu.find(i => i.id === line.id);
        const mPrice = line.mods.reduce((s, mId) => s + (mods.find(m => m.id === mId)?.price || 0), 0);
        sub += item.price + mPrice;
      });
      const tax = Math.round(sub * 0.0825);
      const tip = Math.round(sub * 0.18);
      const total = sub + tax + tip;

      const mode = document.getElementById('split-mode').value;
      let splitText = `$${(total/100).toFixed(2)} Total`;
      if (mode === 'equal') {
        splitText = `Split 2x: $${(total/2/100).toFixed(2)} / guest`;
      } else if (mode === 'itemized' && t.items.length > 0) {
        splitText = `Itemized: Guest 1 ($${(total/100).toFixed(2)})`;
      }

      document.getElementById('totals-box').innerHTML = `
        <div style="display:flex; justify-content:space-between;"><span>Subtotal:</span><span>$${(sub/100).toFixed(2)}</span></div>
        <div style="display:flex; justify-content:space-between;"><span>Tax (8.25%):</span><span>$${(tax/100).toFixed(2)}</span></div>
        <div style="display:flex; justify-content:space-between;"><span>Tip (18%):</span><span>$${(tip/100).toFixed(2)}</span></div>
        <div style="display:flex; justify-content:space-between; font-size:13px; font-weight:800; color:var(--accent); margin-top:4px;">
          <span>${splitText}</span>
        </div>
      `;
    }

    function renderFloorAndKds() {
      document.getElementById('floor-grid').innerHTML = Object.values(tables).map(t => `
        <div class="t-card ${t.items.length ? 'occ' : ''} ${t.id === activeTable ? 'sel' : ''}"
             onclick="activeTable='${t.id}';renderAll();switchTab('register', document.querySelector('nav button'));">
          <div style="font-weight:800;">${t.name}</div>
          <div style="font-size:11px; color:var(--muted); margin:4px 0;">${t.items.length ? t.items.length + ' Items Open' : 'Available'}</div>
        </div>
      `).join('');

      // Populate merge dropdowns
      const opts = Object.values(tables).map(t => `<option value="${t.id}">${t.name}</option>`).join('');
      document.getElementById('merge-src').innerHTML = opts;
      document.getElementById('merge-dest').innerHTML = opts;

      document.getElementById('nav-k').textContent = kds.length;
      document.getElementById('kds-grid').innerHTML = kds.length === 0 ? '<div style="color:var(--muted);">No active kitchen tickets.</div>' : kds.map((k, idx) => `
        <div class="t-card occ" style="text-align:left;">
          <strong>${k.table}</strong>
          <div style="font-size:11px; color:var(--accent); margin:4px 0;">Ticket #${k.id}</div>
          ${k.items.map(i => `<div style="font-size:12px;">• ${i}</div>`).join('')}
          <button class="btn btn-green" style="margin-top:8px; padding:6px;" onclick="kds.splice(${idx},1);renderAll();">Bump Ticket</button>
        </div>
      `).join('');

      document.getElementById('inv-tbody').innerHTML = menu.map(i => `
        <tr>
          <td>${i.id}</td>
          <td><strong>${i.name}</strong></td>
          <td>${i.tags.join(', ').toUpperCase()}</td>
          <td>$${(i.price/100).toFixed(2)}</td>
          <td>
            <span class="badge ${i.stock < 15 ? 'badge-warn' : 'badge-ok'}">
              ${i.stock} units ${i.stock < 15 ? '(LOW STOCK)' : ''}
            </span>
          </td>
          <td><button class="btn btn-blue" style="width:auto; padding:3px 8px;" onclick="i.stock+=10;renderAll();">+ Restock 10</button></td>
        </tr>
      `).join('');
    }

    function executeMerge() {
      const src = document.getElementById('merge-src').value;
      const dest = document.getElementById('merge-dest').value;
      if (src === dest) return alert('Cannot merge a table into itself.');

      tables[dest].items.push(...tables[src].items);
      tables[src].items = [];
      activeTable = dest;
      renderAll();
      alert(`Successfully merged ${tables[src].name} into ${tables[dest].name}!`);
      switchTab('register', document.querySelector('nav button'));
    }

    function chargeAndPrintReceipt() {
      const t = tables[activeTable];
      if (t.items.length === 0) return alert('Check is empty.');
      let sub = 0;
      t.items.forEach(line => {
        const item = menu.find(i => i.id === line.id);
        const mPrice = line.mods.reduce((s, mId) => s + (mods.find(m => m.id === mId)?.price || 0), 0);
        sub += item.price + mPrice;
      });
      const tax = Math.round(sub * 0.0825);
      const tip = Math.round(sub * 0.18);
      const total = sub + tax + tip;
      const id = 'ord_' + Math.random().toString(36).slice(2, 7);

      kds.unshift({
        id: id.toUpperCase(),
        table: t.name,
        items: t.items.map(l => menu.find(i => i.id === l.id).name)
      });

      historyOrders.unshift({
        id,
        channel: 'pos_terminal',
        total,
        stripeFee: Math.round(total * 0.029) + 30,
        compFee: Math.round(total * 0.035) + 15
      });

      // Generate thermal receipt HTML
      lastReceiptHtml = `
        <div>ORDER ID: ${id}</div>
        <div>TABLE: ${t.name}</div>
        <div>TIME: ${new Date().toLocaleTimeString()}</div>
        <div style="border-top:1px dashed #9ca3af; margin:6px 0;"></div>
        ${t.items.map(l => {
          const item = menu.find(i => i.id === l.id);
          return `<div>1x ${item.name} ... $${(item.price/100).toFixed(2)}</div>`;
        }).join('')}
        <div style="border-top:1px dashed #9ca3af; margin:6px 0;"></div>
        <div>SUBTOTAL: $${(sub/100).toFixed(2)}</div>
        <div>TAX (8.25%): $${(tax/100).toFixed(2)}</div>
        <div>TIP (18%): $${(tip/100).toFixed(2)}</div>
        <div style="font-weight:bold; font-size:13px; margin-top:4px;">TOTAL PAID: $${(total/100).toFixed(2)}</div>
        <div style="text-align:center; margin-top:10px; font-size:10px;">Thank you for dining with us!</div>
      `;

      document.getElementById('receipt-body').innerHTML = lastReceiptHtml;
      document.getElementById('receipt-modal').classList.remove('hidden');

      t.items = [];
      renderAll();
    }

    async function printViaBluetooth() {
      if (navigator.bluetooth) {
        try {
          const device = await navigator.bluetooth.requestDevice({ acceptAllDevices: true });
          const server = await device.gatt.connect();
          alert('Connected to Bluetooth thermal printer! Transmitting ESC/POS bytes...');
          document.getElementById('receipt-modal').classList.add('hidden');
          return;
        } catch (e) {
          console.log('Bluetooth canceled or unavailable:', e);
        }
      }
      // Fallback to browser print thermal layout
      window.print();
      document.getElementById('receipt-modal').classList.add('hidden');
    }

    function renderInsights() {
      const totals = historyOrders.reduce((a, o) => {
        a.sales += o.total;
        a.stripe += o.stripeFee;
        a.comp += o.compFee;
        return a;
      }, { sales: 0, stripe: 0, comp: 0 });

      document.getElementById('insights-stats').innerHTML = `
        <div class="mini"><strong>Gross Sales</strong><div style="font-size:16px; font-weight:800; color:var(--accent);">$${(totals.sales/100).toFixed(2)}</div></div>
        <div class="mini"><strong>Stripe Fees</strong><div style="font-size:16px; font-weight:800;">$${(totals.stripe/100).toFixed(2)}</div></div>
        <div class="mini"><strong>Saved w/ Orderly</strong><div style="font-size:16px; font-weight:800; color:var(--accent);">+$${((totals.comp - totals.stripe)/100).toFixed(2)}</div></div>
      `;

      document.getElementById('orders-tbody').innerHTML = historyOrders.map(o => `
        <tr>
          <td>${o.id}</td>
          <td>${o.channel}</td>
          <td>$${(o.total/100).toFixed(2)}</td>
          <td>$${(o.stripeFee/100).toFixed(2)}</td>
          <td>$${(o.compFee/100).toFixed(2)}</td>
          <td style="color:var(--accent); font-weight:700;">+$${((o.compFee - o.stripeFee)/100).toFixed(2)}</td>
        </tr>
      `).join('');
    }

    function switchTab(tab, btn) {
      document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));
      if (btn) btn.classList.add('active');
      ['register', 'floor', 'kds', 'inventory', 'insights'].forEach(v => {
        document.getElementById('tab-' + v).classList.toggle('hidden', v !== tab);
      });
      if (tab === 'insights') renderInsights();
      if (tab === 'inventory' || tab === 'floor') renderFloorAndKds();
    }

    function renderAll() {
      renderMenu();
      renderCheck();
      renderFloorAndKds();
    }
